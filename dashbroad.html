// 容錯升級版 fetchBusETA async function fetchBusETA() { if (!selectedStopId) return; const container = document.getElementById('eta-container'); const now = new Date(); try { const res = await fetch(`https://data.etabus.gov.hk/v1/transport/kmb/eta/${selectedStopId}/${currentRoute}/1`); if (!res.ok) throw new Error('API 連線失敗'); const json = await res.json(); // 放寬條件：只要有 eta，且允許 2 分鐘內的進站緩衝時間，不強制限制 dir const validList = (json.data || []) .filter(item => item.eta && (new Date(item.eta).getTime() >= (now.getTime() - 120000))) .sort((a, b) => new Date(a.eta) - new Date(b.eta)); // 剔除重複時間班次，取前 3 班 const uniqueEtas = []; const seen = new Set(); for (const item of validList) { if (!seen.has(item.eta)) { seen.add(item.eta); uniqueEtas.push(item); } if (uniqueEtas.length >= 3) break; } if (uniqueEtas.length === 0) { container.innerHTML = `
${currentRoute} 目前此站暫無九巴班次預報

💡 619 為九巴/城巴聯營路線，此時段可能正由城巴提供服務。

若此站為終點落客站，亦不會提供候車時間。

`; document.getElementById('last-updated').innerText = `最後查詢: ${now.toLocaleTimeString('zh-HK', { hour12: false })}`; return; } container.innerHTML = uniqueEtas.map((bus, idx) => { const etaDate = new Date(bus.eta); const diffMins = Math.max(0, Math.round((etaDate - now) / 60000)); const isArriving = diffMins <= 3; return `
第 ${idx + 1} 班 (${bus.route}) ${bus.rmk_tc || '九巴實時'}
${diffMins} 分鐘
往: ${bus.dest_tc || '--'} 預計 ${etaDate.toLocaleTimeString('zh-HK', { hour: '2-digit', minute: '2-digit' })}
`; }).join(''); document.getElementById('last-updated').innerText = `最後更新: ${now.toLocaleTimeString('zh-HK', { hour12: false })}`; } catch (err) { console.error('ETA load error:', err); container.innerHTML = `
⚠️ 無法連線至九巴 API
${err.message}
`; } }
